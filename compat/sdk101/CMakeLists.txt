cmake_minimum_required(VERSION 3.22)
project(official_sdk101_audit LANGUAGES C CXX)
foreach(required IN ITEMS SDK_SOURCE PLUGIN_SOURCE REFERENCE_SOURCE CSV_PATH_TEST_SOURCE)
 if(NOT DEFINED ${required} OR NOT EXISTS "${${required}}")
  message(FATAL_ERROR "Set ${required} to an existing exact-source path; see README.md")
 endif()
endforeach()
file(STRINGS "${SDK_SOURCE}/VERSION" audit_sdk_version LIMIT_COUNT 1)
if(NOT audit_sdk_version STREQUAL "1.0.1")
 message(FATAL_ERROR "This audit requires SDK 1.0.1 sources")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(PJ_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PJ_BUILD_PORTED_PLUGINS OFF CACHE BOOL "" FORCE)
set(PJ_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
set(PJ_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(PJ_WARNINGS_AS_ERRORS ON CACHE BOOL "" FORCE)
find_package(GTest REQUIRED)
find_package(nlohmann_json REQUIRED)
add_subdirectory(${SDK_SOURCE} sdk)
set(PJ_WARNING_FLAGS -Wall -Wextra -Werror -Wshadow -Wnon-virtual-dtor -Wold-style-cast -Wcast-qual -Wconversion -Woverloaded-virtual -Wpedantic)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(AUDIT_PLUGINS "data_load_csv;parser_json;toolbox_fft;toolbox_quaternion;data_stream_dummy;data_load_3d" CACHE STRING "Plugin subset")
foreach(plugin IN LISTS AUDIT_PLUGINS)
 add_subdirectory(${PLUGIN_SOURCE}/${plugin} ${plugin})
endforeach()
if(EXISTS ${PLUGIN_SOURCE}/widget_log)
 add_library(log_viewer_core STATIC ${PLUGIN_SOURCE}/widget_log/log_filter.cpp ${PLUGIN_SOURCE}/widget_log/log_loader.cpp ${PLUGIN_SOURCE}/widget_log/log_store.cpp)
 target_compile_options(log_viewer_core PRIVATE ${PJ_WARNING_FLAGS})
 target_include_directories(log_viewer_core PUBLIC ${PLUGIN_SOURCE}/widget_log)
 target_link_libraries(log_viewer_core PUBLIC plotjuggler_sdk::plugin_sdk)
 add_executable(log_viewer_core_test ${PLUGIN_SOURCE}/widget_log/tests/log_core_test.cpp)
 target_compile_options(log_viewer_core_test PRIVATE ${PJ_WARNING_FLAGS})
 target_include_directories(log_viewer_core_test PRIVATE ${PLUGIN_SOURCE}/widget_log/tests)
 target_link_libraries(log_viewer_core_test PRIVATE log_viewer_core plotjuggler_sdk::plugin_host GTest::gtest_main)
 add_test(NAME log_viewer_core_test COMMAND log_viewer_core_test)
endif()

if(NOT TARGET csv_source_test)
 add_executable(csv_source_test ${REFERENCE_SOURCE}/data_load_csv/tests/csv_source_test.cpp)
 target_compile_definitions(csv_source_test PRIVATE CSV_SOURCE_PLUGIN_PATH="$<TARGET_FILE:csv_source_plugin>")
 target_link_libraries(csv_source_test PRIVATE plotjuggler_sdk::plugin_host plotjuggler_sdk::plugin_sdk GTest::gtest_main)
 add_test(NAME csv_source_test COMMAND csv_source_test)
endif()
if(NOT TARGET csv_file_paths_test)
 add_executable(csv_file_paths_test ${CSV_PATH_TEST_SOURCE})
 target_compile_definitions(csv_file_paths_test PRIVATE PJ_CSV_PLUGIN_PATH="$<TARGET_FILE:csv_source_plugin>")
 target_link_libraries(csv_file_paths_test PRIVATE plotjuggler_sdk::plugin_host plotjuggler_sdk::plugin_sdk nlohmann_json::nlohmann_json GTest::gtest_main)
 add_test(NAME csv_file_paths_test COMMAND csv_file_paths_test)
endif()

if(EXISTS ${PLUGIN_SOURCE}/widget_log AND NOT PLUGIN_SOURCE STREQUAL "${REFERENCE_SOURCE}")
 add_executable(log_viewer_391_regressions ${REFERENCE_SOURCE}/widget_log/tests/log_core_test.cpp)
 target_include_directories(log_viewer_391_regressions PRIVATE ${REFERENCE_SOURCE}/widget_log/tests)
 target_link_libraries(log_viewer_391_regressions PRIVATE log_viewer_core plotjuggler_sdk::plugin_host GTest::gtest_main)
 add_test(NAME log_viewer_391_regressions COMMAND log_viewer_391_regressions)
endif()
